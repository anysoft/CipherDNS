include(cmake/LinkMode.cmake)

add_library(cdns_options INTERFACE)
target_include_directories(cdns_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(cdns_options INTERFACE
  MBEDTLS_CONFIG_FILE="cipherdns/mbedtls_config.h"
  CDNS_VERSION="${CIPHERDNS_VERSION}")
if(NOT CIPHERDNS_MODE STREQUAL "CLIENT")
  target_compile_definitions(cdns_options INTERFACE CDNS_SERVER=1)
endif()
if(NOT CIPHERDNS_MODE STREQUAL "SERVER")
  target_compile_definitions(cdns_options INTERFACE CDNS_CLIENT=1)
endif()
if(WIN32)
  target_compile_definitions(cdns_options INTERFACE _WIN32_WINNT=0x0602 WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_definitions(cdns_options INTERFACE _POSIX_C_SOURCE=200809L)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(cdns_options INTERFACE _DEFAULT_SOURCE)
  elseif(APPLE)
    target_compile_definitions(cdns_options INTERFACE _DARWIN_C_SOURCE)
  endif()
endif()

include(cmake/MSVCAtomics.cmake)
cdns_enable_msvc_atomics(cdns_options)

# Sources are checked into the repository. Configuration never downloads dependencies.
file(GLOB mbed_sources CONFIGURE_DEPENDS "third_party/mbedtls/library/*.c")
if(NOT mbed_sources OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/inih/ini.c")
  message(FATAL_ERROR "Vendored dependencies are missing; use a complete source checkout/archive")
endif()
set(mbed_crypto_sources)
set(mbed_tls_sources)
set(mbed_x509_sources)
foreach(source IN LISTS mbed_sources)
  if(source MATCHES "/(ssl[^/]*|mps_[^/]*|debug|net_sockets|timing)\\.c$")
    list(APPEND mbed_tls_sources "${source}")
  elseif(source MATCHES "/(x509[^/]*|pkcs7)\\.c$")
    list(APPEND mbed_x509_sources "${source}")
  else()
    list(APPEND mbed_crypto_sources "${source}")
  endif()
endforeach()
add_library(mbedcrypto STATIC ${mbed_crypto_sources})
target_include_directories(mbedcrypto SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mbedtls/include")
target_include_directories(mbedcrypto PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mbedtls/library")
target_link_libraries(mbedcrypto PUBLIC cdns_options)
if(NOT CIPHERDNS_MODE STREQUAL "CLIENT")
  add_library(mbedx509 STATIC ${mbed_x509_sources})
  target_include_directories(mbedx509 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mbedtls/library")
  target_link_libraries(mbedx509 PUBLIC mbedcrypto)
  add_library(mbedtls STATIC ${mbed_tls_sources})
  target_include_directories(mbedtls PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mbedtls/library")
  target_link_libraries(mbedtls PUBLIC mbedx509 mbedcrypto)
endif()
add_library(inih STATIC third_party/inih/ini.c)
target_include_directories(inih SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/third_party/inih")
target_compile_definitions(inih PRIVATE INI_MAX_LINE=2048 INI_ALLOW_MULTILINE=0 INI_ALLOW_INLINE_COMMENTS=0 INI_STOP_ON_FIRST_ERROR=1)

function(cdns_project_target target)
  target_link_libraries(${target} PUBLIC cdns_options)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
    if(CIPHERDNS_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(CIPHERDNS_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
    target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
  endif()
endfunction()

endif()

  add_library(cipherdns_server STATIC src/ip.c src/http.c src/provider.c src/server.c)
  target_link_libraries(cipherdns_server PUBLIC cipherdns_core mbedtls)
  if(WIN32)
    target_link_libraries(cipherdns_server PUBLIC crypt32)
  endif()
  cdns_project_target(cipherdns_server)

  add_library(cipherdns_client STATIC src/client.c src/dns.c)
  target_link_libraries(cipherdns_client PUBLIC cipherdns_core)
  if(WIN32)
    target_link_libraries(cipherdns_client PUBLIC dnsapi)
  else()
    target_link_libraries(cipherdns_client PUBLIC resolv)
  endif()
  cdns_project_target(cipherdns_client)
endif()
add_executable(cipherdns src/main.c src/doctor.c)
cdns_project_target(cipherdns)
if(TARGET cipherdns_server)
  target_link_libraries(cipherdns PRIVATE cipherdns_server)
endif()
if(TARGET cipherdns_client)
  target_link_libraries(cipherdns PRIVATE cipherdns_client)
endif()

cdns_verify_binary(cipherdns)

if(CIPHERDNS_BUILD_TESTS)
  enable_testing()
  function(cdns_test name)
    add_executable(test_${name} tests/test_${name}.c ${ARGN})
    cdns_project_target(test_${name})
    if(name STREQUAL "client" OR name STREQUAL "reliability")
      target_link_libraries(test_${name} PRIVATE cipherdns_client)
    elseif(name STREQUAL "e2e")
      target_link_libraries(test_${name} PRIVATE cipherdns_server cipherdns_client)
    elseif(NOT name STREQUAL "doctor")
      target_link_libraries(test_${name} PRIVATE cipherdns_core)
    endif()
    # Tests must retain assertions in Release and MinSizeRel builds.
    if(MSVC)
      target_compile_options(test_${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
    endif()
    cdns_link_executable(test_${name})
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 45)
  endfunction()
  cdns_test(core)
  cdns_test(doctor src/doctor.c)
  if(TARGET cipherdns_server)
    target_link_libraries(test_doctor PRIVATE cipherdns_server)
  endif()
  if(TARGET cipherdns_client)
    target_link_libraries(test_doctor PRIVATE cipherdns_client)
  endif()
  if(TARGET cipherdns_server)
    cdns_test(provider src/provider.c)
    cdns_test(server src/server.c src/ip.c)
    cdns_test(http src/http.c)
    target_link_libraries(test_http PRIVATE mbedtls)
    if(WIN32)
      target_link_libraries(test_http PRIVATE crypt32)
    endif()
  endif()
  if(TARGET cipherdns_client)
    cdns_test(client)
    cdns_test(reliability)
    set_tests_properties(reliability PROPERTIES TIMEOUT 120 LABELS reliability)
  endif()
  if(CIPHERDNS_MODE STREQUAL "BOTH")
    cdns_test(e2e)
  endif()
endif()
